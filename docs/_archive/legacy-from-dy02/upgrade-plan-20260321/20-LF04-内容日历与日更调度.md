# 升级项 LF-04: 内容日历 + 日更脚本调度

> **优先级**: P1 中 | **复杂度**: M | **前置**: LF-02（短视频内容类型扩展）
> **改动范围**: 后端新建 Entity/Service/Controller + SQL + 前端 ContentCalendarPage 增强

---

## 问题

日更 5 条短视频缺乏**规划视角**：运营每天零散创建脚本，没有"本周内容排布"的日历视图，无法确保每天的内容类型多样化（不能连续 3 天都发测评）。`ContentCalendarPage.tsx` 已有月视图 UI，但后端没有独立的 `sv_content_calendar` 实体来管理每日内容规划。

-- sv_content_calendar: 每日内容排布规划表
CREATE TABLE IF NOT EXISTS sv_content_calendar (
    id              BIGSERIAL PRIMARY KEY,
    owner_id        BIGINT NOT NULL,
    persona_id      BIGINT,
    plan_date       DATE NOT NULL,
    content_type    VARCHAR(32) NOT NULL,
    title           VARCHAR(256),
    brief           TEXT,
    script_id       BIGINT,
    project_id      BIGINT,
    shooting_task_id BIGINT,
    status          INT NOT NULL DEFAULT 0,
    priority        INT NOT NULL DEFAULT 0,
    publish_time    VARCHAR(16),
    account_id      BIGINT,
    tags            TEXT,
    deleted         INT NOT NULL DEFAULT 0,
    create_time     TIMESTAMP DEFAULT now(),
    update_time     TIMESTAMP DEFAULT now()
);

COMMENT ON TABLE sv_content_calendar IS '短视频内容日历：每日内容排布规划';
COMMENT ON COLUMN sv_content_calendar.content_type IS
  '内容类型：viral_clone/daily/soft_ad/review/tutorial/unboxing/seeding/skit/vlog/before_after/live_preview/trending';
COMMENT ON COLUMN sv_content_calendar.status IS
  '状态：0=计划中, 1=脚本已生成, 2=拍摄中, 3=已拍摄, 4=已发布';
COMMENT ON COLUMN sv_content_calendar.publish_time IS '计划发布时间 HH:mm';

CREATE INDEX IF NOT EXISTS idx_sv_content_cal_owner ON sv_content_calendar(owner_id, deleted, plan_date DESC);
CREATE INDEX IF NOT EXISTS idx_sv_content_cal_date ON sv_content_calendar(plan_date, status) WHERE deleted = 0;
CREATE INDEX IF NOT EXISTS idx_sv_content_cal_persona ON sv_content_calendar(persona_id, plan_date) WHERE deleted = 0 AND persona_id IS NOT NULL;
```

package cn.gaifan.douyinOperations.module.shortvideo.entity;

import jakarta.persistence.*;
import lombok.Data;
import org.hibernate.annotations.SQLRestriction;

import java.time.LocalDate;
import java.time.LocalDateTime;

@Data
@Entity
@Table(name = "sv_content_calendar")
@SQLRestriction("deleted = 0")
public class SvContentCalendar {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(name = "owner_id", nullable = false)
    private Long ownerId;

    @Column(name = "persona_id")
    private Long personaId;

    @Column(name = "plan_date", nullable = false)
    private LocalDate planDate;

    /** 内容类型：viral_clone/daily/soft_ad/review/tutorial/unboxing/seeding/skit/vlog/before_after/live_preview/trending */
    @Column(name = "content_type", length = 32, nullable = false)
    private String contentType;

    @Column(name = "title", length = 256)
    private String title;

    @Column(name = "brief", columnDefinition = "TEXT")
    private String brief;

    @Column(name = "script_id")
    private Long scriptId;

    @Column(name = "project_id")
    private Long projectId;

    @Column(name = "shooting_task_id")
    private Long shootingTaskId;

    /** 状态：0=计划中, 1=脚本已生成, 2=拍摄中, 3=已拍摄, 4=已发布 */
    @Column(name = "status", nullable = false)
    private Integer status = 0;

    @Column(name = "priority")
    private Integer priority = 0;

    /** 计划发布时间 HH:mm */
    @Column(name = "publish_time", length = 16)
    private String publishTime;

    @Column(name = "account_id")
    private Long accountId;

    @Column(name = "tags", columnDefinition = "TEXT")
    private String tags;

    @Column(name = "deleted")
    private Integer deleted = 0;

    @Column(name = "create_time")
    private LocalDateTime createTime;

    @Column(name = "update_time")
    private LocalDateTime updateTime;

    @PrePersist
    protected void onCreate() {
        createTime = LocalDateTime.now();
        updateTime = LocalDateTime.now();
    }

    @PreUpdate
    protected void onUpdate() {
        updateTime = LocalDateTime.now();
    }
}
```

#### `SvContentCalendarServiceImpl.java` — 标准 CRUD 实现

### 6. Controller

```java
@RestController
@RequestMapping("/api/v1/short-video/content-calendar")
@RequiredArgsConstructor
public class SvContentCalendarController {

    private final SvContentCalendarService calendarService;

    @PostMapping("/list")
    public RESTResult<PageResultVO<SvContentCalendarVO>> list(
            @RequestBody SvContentCalendarSearchVO searchVO, @CurrentUserId Long userId) {
        return RESTResult.success(calendarService.list(searchVO, userId));
    }

    @PostMapping("/get")
    public RESTResult<SvContentCalendarVO> get(@RequestBody Map<String, Long> body, @CurrentUserId Long userId) {
        return RESTResult.success(calendarService.get(body.get("id"), userId));
    }

    @PostMapping("/save")
    public RESTResult<Long> save(@Valid @RequestBody SvContentCalendarSaveVO saveVO, @CurrentUserId Long userId) {
        return RESTResult.success(calendarService.save(saveVO, userId));
    }

    @PostMapping("/delete")
    public RESTResult<Void> delete(@RequestBody Map<String, Long> body, @CurrentUserId Long userId) {
        calendarService.delete(body.get("id"), userId);
        return RESTResult.success(null);
    }

    @PostMapping("/date-range")
    public RESTResult<List<SvContentCalendarVO>> dateRange(@RequestBody Map<String, Object> body, @CurrentUserId Long userId) {
        String from = (String) body.get("from");
        String to = (String) body.get("to");
        Long personaId = body.containsKey("personaId") && body.get("personaId") != null
                ? ((Number) body.get("personaId")).longValue() : null;
        return RESTResult.success(calendarService.listByDateRange(from, to, personaId, userId));
    }

    @PostMapping("/auto-generate")
    public RESTResult<Integer> autoGenerate(@RequestBody Map<String, Object> body, @CurrentUserId Long userId) {
        Long personaId = ((Number) body.get("personaId")).longValue();
        String from = (String) body.get("from");
        String to = (String) body.get("to");
        return RESTResult.success(calendarService.autoGenerate(personaId, from, to, userId));
    }
}
```

### 7. 前端 API: `frontend-react/src/api/content-calendar.ts` — 新建

```typescript
import request from '@/utils/request'

const BASE = '/short-video/content-calendar'

export interface ContentCalendarItem {
  id: number
  personaId?: number
  planDate: string
  contentType: string
  title?: string
  brief?: string
  scriptId?: number
  projectId?: number
  shootingTaskId?: number
  status: number
  priority: number
  publishTime?: string
  accountId?: number
  tags?: string
}

export function listContentCalendar(params: Record<string, unknown>) {
  return request.post<{ total: number; list: ContentCalendarItem[] }>(`${BASE}/list`, params)
}

export function getContentCalendar(id: number) {
  return request.post<ContentCalendarItem>(`${BASE}/get`, { id })
}

export function saveContentCalendar(data: Partial<ContentCalendarItem>) {
  return request.post<number>(`${BASE}/save`, data)
}

export function deleteContentCalendar(id: number) {
  return request.post(`${BASE}/delete`, { id })
}

export function listByDateRange(from: string, to: string, personaId?: number) {
  return request.post<ContentCalendarItem[]>(`${BASE}/date-range`, { from, to, personaId })
}

export function autoGenerateCalendar(personaId: number, from: string, to: string) {
  return request.post<number>(`${BASE}/auto-generate`, { personaId, from, to })
}
```
