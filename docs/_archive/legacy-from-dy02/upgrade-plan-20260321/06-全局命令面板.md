# D1-01: 全局命令面板 (Ctrl+K)

> **优先级**: P0 | **复杂度**: M | **维度**: D1 导航与信息架构
> **前置**: 无 | **改动文件**: 新建 `CommandPalette.tsx`, 修改 `BaseLayout.tsx`

---

## 问题

50+ 菜单项，找一个功能需要 3+ 次点击和滚动。

---

## 方案

新建 CommandPalette 组件，基于 MUI Dialog + Autocomplete：
- `Ctrl+K` / `Cmd+K` 快速打开
- 数据源 = 路由表（router/index.tsx 中的路由配置）+ 最近访问
- 模糊搜索中文 + 拼音首字母
- 回车或点击跳转

---

### 1. `frontend-react/src/components/base/CommandPalette.tsx` — 新建

```tsx
import { useState, useEffect, useMemo, useCallback } from 'react'
import { useNavigate } from 'react-router-dom'
import {
  Dialog,
  TextField,
  List,
  ListItemButton,
  ListItemIcon,
  ListItemText,
  InputAdornment,
  Typography,
  Box,
  Chip,
} from '@mui/material'
import SearchIcon from '@mui/icons-material/Search'
import { useRolePrefix } from '@/hooks/useRolePrefix'

interface CommandItem {
  label: string
  path: string
  keywords: string[]    // 搜索关键词（含拼音首字母）
  category: string
  icon?: React.ReactNode
}

// 完整的路由→标签映射表
const ROUTE_LABELS: Array<{ path: string; label: string; category: string; keywords: string[] }> = [
  // 直播

  { path: '/live/product', label: '直播选品', category: '直播', keywords: ['xuanpin', 'product'] },
  { path: '/live/compare', label: '场次对比', category: '直播', keywords: ['duibi', 'compare'] },
  { path: '/live/script-versions', label: '话术版本', category: '直播', keywords: ['huashu', 'banben', 'script', 'version'] },
  // 商品
  { path: '/product', label: '商品管理', category: '商品', keywords: ['shangpin', 'product'] },
  { path: '/product/readiness', label: '商品就绪度', category: '商品', keywords: ['jiuxu', 'readiness'] },
  { path: '/product/effectiveness', label: '效果评分', category: '商品', keywords: ['xiaoguo', 'effectiveness'] },
  { path: '/product/style-preset', label: '风格预设', category: '商品', keywords: ['fengge', 'style'] },
  // 话术
  { path: '/script/list', label: '话术管理', category: '话术', keywords: ['huashu', 'script'] },
  { path: '/script/templates', label: '话术模板', category: '话术', keywords: ['moban', 'template'] },

  { path: '/shortvideo/dashboard', label: '短视频工作台', category: '短视频', keywords: ['duanshipin', 'shortvideo', 'dashboard'] },
  { path: '/shortvideo/script-planning', label: '脚本策划', category: '短视频', keywords: ['jiaoben', 'cehua', 'script', 'planning'] },

  { path: '/config', label: '配置管理', category: '系统', keywords: ['peizhi', 'config'] },
  { path: '/system', label: '系统监控', category: '系统', keywords: ['xitong', 'system', 'monitor'] },

interface CommandPaletteProps {
  open: boolean
  onClose: () => void
}

export function CommandPalette({ open, onClose }: CommandPaletteProps) {
  const navigate = useNavigate()
  const prefix = useRolePrefix()
  const [query, setQuery] = useState('')
  const [selectedIndex, setSelectedIndex] = useState(0)

  // 搜索过滤
  const filtered = useMemo(() => {
    if (!query.trim()) return ROUTE_LABELS.slice(0, 15) // 默认显示前15项
    const q = query.toLowerCase().trim()
    return ROUTE_LABELS.filter((item) =>
      item.label.toLowerCase().includes(q) ||
      item.category.toLowerCase().includes(q) ||
      item.keywords.some((kw) => kw.includes(q))
    ).slice(0, 15)
  }, [query])

  // 重置选择
  useEffect(() => { setSelectedIndex(0) }, [query])

  // 打开时重置
  useEffect(() => {
    if (open) { setQuery(''); setSelectedIndex(0) }
  }, [open])

  const handleSelect = useCallback((item: typeof ROUTE_LABELS[number]) => {
    navigate(`${prefix}${item.path}`)
    onClose()
  }, [navigate, prefix, onClose])

  // 键盘导航
  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === 'ArrowDown') {
      e.preventDefault()
      setSelectedIndex((i) => Math.min(i + 1, filtered.length - 1))
    } else if (e.key === 'ArrowUp') {
      e.preventDefault()
      setSelectedIndex((i) => Math.max(i - 1, 0))
    } else if (e.key === 'Enter' && filtered[selectedIndex]) {
      handleSelect(filtered[selectedIndex])
    }
  }

  return (
    <Dialog
      open={open}
      onClose={onClose}
      maxWidth="sm"
      fullWidth
      PaperProps={{
        sx: {
          position: 'fixed',
          top: '15%',
          m: 0,
          maxHeight: '60vh',
          borderRadius: 2,
        },
      }}
    >
      <TextField
        autoFocus
        fullWidth
        placeholder="搜索页面... (输入中文或拼音)"
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        onKeyDown={handleKeyDown}
        InputProps={{
          startAdornment: (
            <InputAdornment position="start">
              <SearchIcon color="action" />
            </InputAdornment>
          ),
          endAdornment: (
            <InputAdornment position="end">
              <Typography variant="caption" color="text.disabled">ESC 关闭</Typography>
            </InputAdornment>
          ),
        }}
        sx={{
          '& .MuiOutlinedInput-root': {
            '& fieldset': { border: 'none' },
          },
          borderBottom: '1px solid',
          borderColor: 'divider',
        }}
      />
      <List sx={{ overflow: 'auto', py: 0.5 }}>
        {filtered.length === 0 ? (
          <Box sx={{ py: 3, textAlign: 'center' }}>
            <Typography color="text.secondary">没有匹配的页面</Typography>
          </Box>
        ) : (
          filtered.map((item, index) => (
            <ListItemButton
              key={item.path}
              selected={index === selectedIndex}
              onClick={() => handleSelect(item)}
              sx={{ py: 0.75, px: 2 }}
            >
              <ListItemText
                primary={item.label}
                secondary={`${prefix}${item.path}`}
                primaryTypographyProps={{ variant: 'body2', fontWeight: index === selectedIndex ? 600 : 400 }}
                secondaryTypographyProps={{ variant: 'caption' }}
              />
              <Chip size="small" label={item.category} variant="outlined"
                sx={{ height: 20, '& .MuiChip-label': { px: 1, fontSize: '0.7rem' } }} />
            </ListItemButton>
          ))
        )}
      </List>
    </Dialog>
  )
}
```

### 2. `frontend-react/src/layouts/BaseLayout.tsx` — 注册快捷键

在 BaseLayout 组件中：

```tsx
import { CommandPalette } from '@/components/base/CommandPalette'

// 在 BaseLayout 组件内部
const [commandPaletteOpen, setCommandPaletteOpen] = useState(false)

// 注册 Ctrl+K / Cmd+K 快捷键
useEffect(() => {
  const handler = (e: KeyboardEvent) => {
    if ((e.ctrlKey || e.metaKey) && e.key === 'k') {
      e.preventDefault()
      setCommandPaletteOpen(true)
    }
  }
  window.addEventListener('keydown', handler)
  return () => window.removeEventListener('keydown', handler)
}, [])

// 在 JSX 末尾（</Box> 之前）渲染
<CommandPalette open={commandPaletteOpen} onClose={() => setCommandPaletteOpen(false)} />
```

在 AppBar 中增加搜索按钮（可选，方便移动端用户）：

```tsx
{/* 在 AppBar 的按钮区域 */}
<Tooltip title="搜索页面 (Ctrl+K)">
  <IconButton size="small" onClick={() => setCommandPaletteOpen(true)}>
    <SearchIcon />
  </IconButton>
</Tooltip>
```

## 注意事项

- `useRolePrefix` 已在项目中使用，路径前缀如 `/admin` / `/org` / `/talent`
- ROUTE_LABELS 中的 path 不含 prefix，select 时拼接
- Dialog 固定在页面顶部 15% 位置，类似 VS Code 命令面板风格
- 搜索框用 TextField 而非 Autocomplete，方便自定义键盘导航
- 需要导入 `SearchIcon` from `@mui/icons-material/Search`
