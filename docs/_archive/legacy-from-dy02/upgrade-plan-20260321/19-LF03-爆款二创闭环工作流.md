# 升级项 LF-03: 爆款二创闭环工作流

> **优先级**: P0 高 | **复杂度**: L | **前置**: 无（可与 LF-01/LF-02 并行）
> **改动范围**: 后端 Entity/SQL + ViralVideoService + 新增 ViralRemakeService + 前端 ViralLibraryPage 增强

---

## 问题

`SvViralVideo` 检测到爆款后**断裂**：没有 `remake_status` 字段跟踪二创进度，没有 `remake_suggestions` 存储 AI 推荐的二创方向，没有 `industry_tags` 做行业过滤，没有从"发现爆款→运营确认→生成脚本→分配拍摄→发布"的完整闭环。当前 `replicateViral()` 只生成文本返回，不落库不追踪。

二创状态机按 `11-方向基线与防漂移.md` §2.3 冻结：`0未处理 → 1已推荐 → 2运营已确认 → 3脚本已生成 → 4已进拍摄 → 5已完成`。

---

### 4. 新建 `ViralRemakeServiceImpl.java` — 实现

```java
package cn.gaifan.douyinOperations.module.shortvideo.service.impl;

import cn.gaifan.douyinOperations.common.constant.ErrorCode;
import cn.gaifan.douyinOperations.common.exception.BusinessException;
import cn.gaifan.douyinOperations.module.shortvideo.entity.SvViralVideo;
import cn.gaifan.douyinOperations.module.shortvideo.repository.SvViralVideoRepository;
import cn.gaifan.douyinOperations.module.shortvideo.service.ViralRemakeService;
import cn.gaifan.douyinOperations.module.shortvideo.service.ViralVideoService;
import cn.gaifan.douyinOperations.module.ai.service.LlmChatService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;

@Slf4j
@Service
@RequiredArgsConstructor
public class ViralRemakeServiceImpl implements ViralRemakeService {

    private final SvViralVideoRepository viralVideoRepository;

    @org.springframework.beans.factory.annotation.Autowired(required = false)
    private LlmChatService llmChatService;

    @org.springframework.beans.factory.annotation.Autowired(required = false)
    private cn.gaifan.douyinOperations.module.shortvideo.service.ShortVideoAiService shortVideoAiService;

    @org.springframework.beans.factory.annotation.Autowired(required = false)
    private cn.gaifan.douyinOperations.module.shortvideo.service.SvShootingTaskService shootingTaskService;

    @Override
    @Transactional
    public void recommendRemake(Long viralVideoId, Long userId) {
        SvViralVideo viral = viralVideoRepository.findById(viralVideoId)
                .orElseThrow(() -> new BusinessException(ErrorCode.DATA_NOT_FOUND, "爆款视频不存在"));

        if (viral.getRemakeStatus() != null && viral.getRemakeStatus() > 0) {
            throw new BusinessException(ErrorCode.VALIDATION_FAIL, "该视频已有二创推荐");
        }

        // AI 分析生成二创建议
        String suggestions = generateRemakeSuggestions(viral);
        String matchedPersonas = matchPersonas(viral);
        String industryTags = extractIndustryTags(viral);

        viral.setRemakeStatus(1);
        viral.setRemakeSuggestions(suggestions);
        viral.setMatchedPersonaIds(matchedPersonas);
        viral.setIndustryTags(industryTags);
        viralVideoRepository.save(viral);

        log.info("[二创推荐] viralId={}, userId={}", viralVideoId, userId);
    }

    @Override
    @Transactional
    public int batchRecommend(Long userId, double scoreThreshold, int limit) {
        // 查询 viralScore >= threshold 且 remake_status=0 的视频
        List<SvViralVideo> candidates = viralVideoRepository.findAll((root, query, cb) -> {
            query.orderBy(cb.desc(root.get("viralScore")));
            return cb.and(
                    cb.greaterThanOrEqualTo(root.get("viralScore"), scoreThreshold),
                    cb.or(cb.isNull(root.get("remakeStatus")), cb.equal(root.get("remakeStatus"), 0))
            );
        }).stream().limit(limit).toList();

        int count = 0;
        for (SvViralVideo viral : candidates) {
            try {
                recommendRemake(viral.getId(), userId);
                count++;
            } catch (Exception e) {
                log.warn("[二创批量推荐] 跳过 viralId={}: {}", viral.getId(), e.getMessage());
            }
        }
        return count;
    }

    @Override
    @Transactional
    public void confirmRemake(Long viralVideoId, String selectedRemakeType, Long personaId, Long userId) {
        SvViralVideo viral = viralVideoRepository.findById(viralVideoId)
                .orElseThrow(() -> new BusinessException(ErrorCode.DATA_NOT_FOUND));

        if (viral.getRemakeStatus() == null || viral.getRemakeStatus() != 1) {
            throw new BusinessException(ErrorCode.VALIDATION_FAIL, "当前状态不可确认，需先 AI 推荐");
        }

        viral.setRemakeStatus(2);
        viral.setConfirmedBy(userId);
        viral.setConfirmedAt(LocalDateTime.now());
        // 如果运营选择了特定人设，更新 matchedPersonaIds
        if (personaId != null) {
            viral.setMatchedPersonaIds("[" + personaId + "]");
        }
        viralVideoRepository.save(viral);
    }

    @Override
    @Transactional
    public Long generateRemakeScript(Long viralVideoId, Long userId) {
        SvViralVideo viral = viralVideoRepository.findById(viralVideoId)
                .orElseThrow(() -> new BusinessException(ErrorCode.DATA_NOT_FOUND));

        if (viral.getRemakeStatus() == null || viral.getRemakeStatus() != 2) {
            throw new BusinessException(ErrorCode.VALIDATION_FAIL, "需先运营确认才能生成脚本");
        }

        // 调用现有 replicateViral 或 shortVideoAiService 生成脚本
        // 此处用 ViralVideoService.replicateViral 已有逻辑
        // 需在 replicateViral 返回值中拿到 scriptId
        // MVP: 直接创建 SvScript 并关联
        Long scriptId = createRemakeScript(viral, userId);

        viral.setRemakeStatus(3);
        viral.setRemakeScriptId(scriptId);
        viralVideoRepository.save(viral);

        return scriptId;
    }

    @Override
    @Transactional
    public Long assignToShootingTask(Long viralVideoId, Long photographerId, String shootDate, Long userId) {
        SvViralVideo viral = viralVideoRepository.findById(viralVideoId)
                .orElseThrow(() -> new BusinessException(ErrorCode.DATA_NOT_FOUND));

        if (viral.getRemakeStatus() == null || viral.getRemakeStatus() != 3) {
            throw new BusinessException(ErrorCode.VALIDATION_FAIL, "需先生成脚本才能分配任务");
        }

        if (shootingTaskService == null) {
            throw new BusinessException(ErrorCode.SYSTEM_ERROR, "拍摄任务服务不可用");
        }

        // 创建拍摄任务
        var taskVo = new cn.gaifan.douyinOperations.module.shortvideo.vo.SvShootingTaskSaveVO();
        taskVo.setTitle("二创: " + (viral.getTitle() != null ? viral.getTitle().substring(0, Math.min(50, viral.getTitle().length())) : "爆款二创"));
        taskVo.setScriptId(viral.getRemakeScriptId());
        taskVo.setPhotographerId(photographerId);
        taskVo.setShootDate(LocalDate.parse(shootDate));
        taskVo.setDescription("基于爆款视频二创，原视频: " + (viral.getVideoUrl() != null ? viral.getVideoUrl() : ""));

        Long taskId = shootingTaskService.save(taskVo, userId);

        viral.setRemakeStatus(4);
        viral.setRemakeTaskId(taskId);
        viralVideoRepository.save(viral);

        return taskId;
    }

    @Override
    @Transactional
    public void markCompleted(Long viralVideoId, Long userId) {
        SvViralVideo viral = viralVideoRepository.findById(viralVideoId)
                .orElseThrow(() -> new BusinessException(ErrorCode.DATA_NOT_FOUND));

        viral.setRemakeStatus(5);
        viralVideoRepository.save(viral);
    }

    // ---- 私有辅助方法 ----

    private String generateRemakeSuggestions(SvViralVideo viral) {
        if (llmChatService == null) {
            // 降级：返回默认建议模板
            return """
                [{"remakeType":"form_imitation","angle":"保持原结构，替换产品","brief":"模仿视频结构和节奏","matchScore":0.8},
                 {"remakeType":"content_flip","angle":"反转原视频观点","brief":"从相反角度解读同一话题","matchScore":0.6},
                 {"remakeType":"element_recombination","angle":"提取爆点重新组合","brief":"保留核心爆点，替换场景和产品","matchScore":0.7}]
                """;
        }

        String prompt = String.format("""

        try {
            return llmChatService.chat(prompt);
        } catch (Exception e) {
            log.warn("[二创建议生成] LLM 调用失败: {}", e.getMessage());

    private String matchPersonas(SvViralVideo viral) {
        // MVP: 返回空数组，二期通过向量相似度匹配人设
        return "[]";
    }

    private String extractIndustryTags(SvViralVideo viral) {
        // 从分析结果或分类推断行业标签
        String category = viral.getCategoryId() != null ? viral.getCategoryId().toString() : "";

        return "[\"护肤\",\"彩妆\"]";
    }

    private Long createRemakeScript(SvViralVideo viral, Long userId) {
        // 创建 SvScript 实体
        var script = new cn.gaifan.douyinOperations.module.shortvideo.entity.SvScript();

        script.setGenerationType("ai");
        script.setReferenceViralId(viral.getId());

            try {
```

### 5. 新建 `ViralRemakeController.java` — API 端点

```java
package cn.gaifan.douyinOperations.module.shortvideo.controller;

import cn.gaifan.douyinOperations.common.annotation.CurrentUserId;
import cn.gaifan.douyinOperations.common.vo.RESTResult;
import cn.gaifan.douyinOperations.module.shortvideo.service.ViralRemakeService;
import lombok.RequiredArgsConstructor;
import org.springframework.web.bind.annotation.*;

import java.util.Map;

@RestController
@RequestMapping("/api/v1/short-video/viral-remake")
@RequiredArgsConstructor
public class ViralRemakeController {

    private final ViralRemakeService viralRemakeService;

    /** AI 推荐二创方向 */
    @PostMapping("/recommend")
    public RESTResult<Void> recommend(@RequestBody Map<String, Object> body, @CurrentUserId Long userId) {
        Long viralVideoId = ((Number) body.get("viralVideoId")).longValue();
        viralRemakeService.recommendRemake(viralVideoId, userId);
        return RESTResult.success(null);
    }

    /** 批量 AI 推荐 */
    @PostMapping("/batch-recommend")
    public RESTResult<Integer> batchRecommend(@RequestBody Map<String, Object> body, @CurrentUserId Long userId) {
        double threshold = body.containsKey("scoreThreshold") ? ((Number) body.get("scoreThreshold")).doubleValue() : 70.0;
        int limit = body.containsKey("limit") ? ((Number) body.get("limit")).intValue() : 20;
        int count = viralRemakeService.batchRecommend(userId, threshold, limit);
        return RESTResult.success(count);
    }

    /** 运营确认二创方向 */
    @PostMapping("/confirm")
    public RESTResult<Void> confirm(@RequestBody Map<String, Object> body, @CurrentUserId Long userId) {
        Long viralVideoId = ((Number) body.get("viralVideoId")).longValue();
        String remakeType = (String) body.get("remakeType");
        Long personaId = body.containsKey("personaId") && body.get("personaId") != null
                ? ((Number) body.get("personaId")).longValue() : null;
        viralRemakeService.confirmRemake(viralVideoId, remakeType, personaId, userId);
        return RESTResult.success(null);
    }

    /** 生成二创脚本 */
    @PostMapping("/generate-script")
    public RESTResult<Long> generateScript(@RequestBody Map<String, Object> body, @CurrentUserId Long userId) {
        Long viralVideoId = ((Number) body.get("viralVideoId")).longValue();
        Long scriptId = viralRemakeService.generateRemakeScript(viralVideoId, userId);
        return RESTResult.success(scriptId);
    }

    /** 分配拍摄任务 */
    @PostMapping("/assign-task")
    public RESTResult<Long> assignTask(@RequestBody Map<String, Object> body, @CurrentUserId Long userId) {
        Long viralVideoId = ((Number) body.get("viralVideoId")).longValue();
        Long photographerId = body.containsKey("photographerId") && body.get("photographerId") != null
                ? ((Number) body.get("photographerId")).longValue() : null;
        String shootDate = (String) body.get("shootDate");
        Long taskId = viralRemakeService.assignToShootingTask(viralVideoId, photographerId, shootDate, userId);
        return RESTResult.success(taskId);
    }

    /** 标记完成 */
    @PostMapping("/complete")
    public RESTResult<Void> complete(@RequestBody Map<String, Object> body, @CurrentUserId Long userId) {
        Long viralVideoId = ((Number) body.get("viralVideoId")).longValue();
        viralRemakeService.markCompleted(viralVideoId, userId);
        return RESTResult.success(null);
    }
}
```

```tsx
  </Button>
)}
{item.remakeStatus === 4 && (
  <Button size="small" onClick={() => handleComplete(item.id)}>
    标记完成
  </Button>
)}
{item.remakeStatus === 5 && (
  <Chip size="small" label="已完成" color="success" />
)}
```
