## 1. 产品概述

### 1.1 现状分析

**核心问题**：
- ❌ 当前只有 REST API，无可视化管理界面
- ❌ 操作需要手动调用接口，门槛高
- ❌ 缺少直观的数据统计和监控
- ❌ 异步任务无进度反馈
- ❌ 批量操作困难

**用户痛点**：
1. **运营人员**：无法直观管理知识库、查看进化报告
2. **内容创作者**：图像/视频生成流程复杂，缺少预览
3. **管理员**：监控数据分散，缺少统一看板

### 1.2 设计目标

**核心价值**：
- ✅ **可视化管理**：所有功能通过 UI 操作，零技术门槛
- ✅ **实时反馈**：异步任务进度可视化，WebSocket 实时推送
- ✅ **数据驱动**：完整的数据看板，支持导出和分析
- ✅ **高效操作**：批量处理、快捷键、智能推荐
- ✅ **体验优先**：流畅交互、优雅动效、响应式设计

**成功指标**：
- 操作效率提升 **80%**（原需 10 次 API 调用 → 1 次点击）
- 学习成本降低 **90%**（无需技术背景）
- 用户满意度 **> 4.5/5**

## 2. 设计原则

### 2.1 核心设计原则

#### 1. **极简主义**（Minimalism）
- **少即是多**：每个页面聚焦单一任务
- **渐进式披露**：复杂功能分步引导
- **空白留足**：避免信息过载

#### 2. **一致性**（Consistency）
- **视觉一致**：统一的颜色、字体、间距
- **交互一致**：相同操作相同位置
- **命名一致**：统一术语体系

#### 3. **即时反馈**（Instant Feedback）
- **操作确认**：每个操作有明确反馈
- **进度可见**：异步任务实时进度
- **状态明确**：加载、成功、失败状态清晰

#### 4. **容错设计**（Error Tolerance）
- **操作可撤销**：关键操作支持撤销
- **二次确认**：危险操作弹窗确认
- **错误提示**：明确告知原因和解决方案

#### 5. **性能优先**（Performance First）
- **首屏 < 1s**：关键资源优先加载
- **分页加载**：大列表虚拟滚动
- **缓存优先**：智能缓存策略

#### 4.1.2 核心功能

- 支持缩放、拖拽、时间范围选择

- 实时显示系统告警
- 颜色编码（⚠️ 警告、🔴 严重、✅ 已恢复）
- 点击跳转监控详情

**4. 快捷操作**
- [导出] 按钮：导出报表（Excel/CSV）
- [刷新] 按钮：手动刷新数据
- [配置] 按钮：自定义显示指标

```tsx
import React from 'react';
```

#### 6.2.2 SearchBar（智能搜索栏）

```tsx
// src/components/SearchBar/index.tsx

import React, { useState } from 'react';
import { Input, Tag, Button } from 'antd';
import { SearchOutlined, AudioOutlined } from '@ant-design/icons';

interface SearchBarProps {
  placeholder?: string;
  hotKeywords?: string[];
  onSearch: (query: string) => void;
  onVoiceSearch?: () => void;
}

export const SearchBar: React.FC<SearchBarProps> = ({
  placeholder = '搜索知识库...',
  hotKeywords = [],
  onSearch,
  onVoiceSearch,
}) => {
  const [query, setQuery] = useState('');

  const handleSearch = () => {
    if (query.trim()) {
      onSearch(query.trim());
    }
  };

  return (
    <div>
      {/* 搜索输入框 */}
      <Input.Search
        size="large"
        placeholder={placeholder}
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        onSearch={handleSearch}
        enterButton={
          <Button type="primary" icon={<SearchOutlined />}>
            搜索
          </Button>
        }
        addonAfter={
          onVoiceSearch && (
            <Button
              type="text"
              icon={<AudioOutlined />}
              onClick={onVoiceSearch}
            />
          )
        }
      />

      {/* 热门搜索 */}
      {hotKeywords.length > 0 && (
        <div style={{ marginTop: 12 }}>
          <span style={{ marginRight: 8, color: '#666' }}>🔥 热门搜索:</span>
          {hotKeywords.map((keyword) => (
            <Tag
              key={keyword}
              style={{ cursor: 'pointer', marginBottom: 8 }}
              onClick={() => {
                setQuery(keyword);
                onSearch(keyword);
              }}
            >
              {keyword}
            </Tag>
          ))}
        </div>
      )}
    </div>
  );
};
```

```typescript
// 响应拦截器
instance.interceptors.response.use(
  (response) => {
    const { data } = response;

    // 统一处理业务错误
    if (data.success === false) {
      message.error(data.message || '请求失败');
      return Promise.reject(new Error(data.message));
    }

    return data;
  },
  (error) => {
    // 统一错误处理
    if (error.response) {
      const { status, data } = error.response;

      switch (status) {
        case 401:
          message.error('未登录或登录已过期');
          // 跳转登录页
          window.location.href = '/login';
          break;
        case 403:
          message.error('无权限访问');
          break;
        case 404:
          message.error('请求的资源不存在');
          break;
        case 500:
          message.error(data?.message || '服务器内部错误');
          break;
        default:
          message.error(data?.message || '请求失败');
      }
    } else if (error.request) {
      message.error('网络错误，请检查您的网络连接');
    } else {
      message.error('请求配置错误');
    }

    return Promise.reject(error);
  }
);

export default instance;
```

```typescript
  deleteKb: (kbId: number) => Promise<void>;

    error: null,

      set({ loading: true, error: null });
      try {
        const newKb = await knowledgeAPI.createKb(data);
        set((state) => ({
          kbList: [newKb, ...state.kbList],
          loading: false,
        }));
      } catch (error: any) {

        throw error;

      set({ loading: true, error: null });
      try {
        await knowledgeAPI.deleteKb(kbId);
        set((state) => ({
          kbList: state.kbList.filter((k) => k.id !== kbId),
          currentKb: state.currentKb?.id === kbId ? null : state.currentKb,
          loading: false,
        }));
      } catch (error: any) {
        set({ error: error.message, loading: false });
        throw error;
      }
    },

    // 上传文档
    uploadDocument: async (kbId: number, data: any) => {
      set({ loading: true, error: null });
      try {
        const doc = await knowledgeAPI.uploadDocument(kbId, data);
        set((state) => ({
          documents: [doc, ...state.documents],
          loading: false,
        }));
      } catch (error: any) {
        set({ error: error.message, loading: false });
        throw error;
      }
    },
  }))
```

#### 7.5.1 useImportProgress.ts（导入进度）

```typescript
// src/hooks/useImportProgress.ts

import { useState, useEffect, useCallback } from 'react';
import { knowledgeAPI } from '@/api/knowledge';
import { useWebSocket } from './useWebSocket';

interface ImportProgress {
  jobId: string;
  status: 'pending' | 'processing' | 'completed' | 'failed';
  progress: number;
  stages: Array<{
    name: string;
    status: 'wait' | 'process' | 'finish' | 'error';
    detail?: string;
  }>;
  estimatedTime?: number;
  error?: string;
}

export const useImportProgress = (jobId: string | null, mode: 'websocket' | 'polling' = 'websocket') => {
  const [progress, setProgress] = useState<ImportProgress | null>(null);
  const [loading, setLoading] = useState(false);

  // WebSocket 模式
  const { message } = useWebSocket(`/import-progress/${jobId}`);

  useEffect(() => {
    if (mode === 'websocket' && message) {
      setProgress(message as ImportProgress);
    }
  }, [message, mode]);

  // 轮询模式
  const pollProgress = useCallback(async () => {
    if (!jobId) return;

    setLoading(true);
    try {
      const data = await knowledgeAPI.getImportStatus(jobId);
      setProgress(data as any);

      // 如果未完成，继续轮询
      if (data.status === 'processing') {
        setTimeout(pollProgress, 2000); // 2秒轮询一次
      }
    } catch (error) {
      console.error('获取导入进度失败:', error);
    } finally {
      setLoading(false);
    }
  }, [jobId]);

  useEffect(() => {
    if (mode === 'polling' && jobId) {
      pollProgress();
    }
  }, [mode, jobId, pollProgress]);

  return { progress, loading };
};
```

#### 7.5.2 useQuotaCheck.ts（额度检查）

```typescript
// src/hooks/useQuotaCheck.ts

import { useState, useEffect } from 'react';
import { request } from '@/utils/request';
import { message } from 'antd';

interface QuotaInfo {
  usedCount: number;
  maxCount: number;
  remaining: number;
  percentage: number;
}

export const useQuotaCheck = () => {
  const [quota, setQuota] = useState<QuotaInfo | null>(null);
  const [loading, setLoading] = useState(false);

  const fetchQuota = async () => {
    setLoading(true);
    try {
      const data = await request.get<any>('/ai/quota/info');
      setQuota({
        usedCount: data.usedCount,
        maxCount: data.maxCount,
        remaining: data.maxCount - data.usedCount,
        percentage: (data.usedCount / data.maxCount) * 100,
      });
    } catch (error) {
      console.error('获取额度信息失败:', error);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchQuota();
  }, []);

  // 检查额度是否充足
  const checkQuota = (required: number = 1): boolean => {
    if (!quota) {
      message.warning('正在获取额度信息...');
      return false;
    }

    if (quota.remaining < required) {
      message.error(`额度不足！需要 ${required} 次，剩余 ${quota.remaining} 次`);
      return false;
    }

    return true;
  };

  // 消耗额度（前端乐观更新）
  const consumeQuota = (count: number = 1) => {
    if (quota) {
      setQuota({
        ...quota,
        usedCount: quota.usedCount + count,
        remaining: quota.remaining - count,
        percentage: ((quota.usedCount + count) / quota.maxCount) * 100,
      });
    }
  };

  return {
    quota,
    loading,
    checkQuota,
    consumeQuota,
    refetch: fetchQuota,
  };
};
```

```typescript
// src/utils/websocket.ts

import { io, Socket } from 'socket.io-client';

class WebSocketManager {
  private socket: Socket | null = null;

  connect(namespace: string = '') {
    if (this.socket?.connected) {
      return this.socket;
    }

    this.socket = io(`ws://localhost:8188${namespace}`, {
      transports: ['websocket'],
      auth: {
        token: localStorage.getItem('token'),
      },
    });

    this.socket.on('connect', () => {
      console.log('WebSocket 已连接');
    });

    this.socket.on('disconnect', () => {
      console.log('WebSocket 已断开');
    });

    this.socket.on('error', (error) => {
      console.error('WebSocket 错误:', error);
    });

    return this.socket;
  }

  disconnect() {
    this.socket?.disconnect();
    this.socket = null;
  }

  // 订阅导入进度
  subscribeImportProgress(jobId: string, callback: (data: any) => void) {
    this.socket?.emit('subscribe-import', { jobId });
    this.socket?.on(`import-progress-${jobId}`, callback);
  }

  // 取消订阅
  unsubscribeImportProgress(jobId: string) {
    this.socket?.emit('unsubscribe-import', { jobId });
    this.socket?.off(`import-progress-${jobId}`);
  }
}

export default new WebSocketManager();
```
