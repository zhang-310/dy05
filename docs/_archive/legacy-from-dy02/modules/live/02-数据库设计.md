# 直播模块 — 数据库设计

> 版本：3.0 | 更新日期：2026-02-26 | 与 00-大纲 对齐（7 张表含 live_script_template）

## 2.1 ER 关系总览

```
dy_account (douyin 模块)
    │
    ├─── live_session（直播场次）
    │        │
    │        ├─── live_session_product（场次-产品关联）
    │        │        │
    │        │        └─── live_product_data（分产品数据）
    │        │
    │        ├─── live_session_script（场次话术）
    │        │
    │        ├─── live_session_data（场次汇总数据）
    │        │
    │        └─── live_monitor（实时时序数据）
    │
    dy_persona (douyin 模块) ──→ live_session.persona_id
    dy_product (douyin 模块) ──→ live_session_product.product_id
    live_session_script ──→ live_script_template（高效话术入库）
```

## 2.3 初始化 SQL

```sql
-- ============================================
-- live 模块数据库初始化脚本
-- ============================================

-- 1. live_session（直播场次）
CREATE TABLE IF NOT EXISTS live_session (
    id BIGSERIAL PRIMARY KEY,
    owner_id BIGINT NOT NULL,
    account_id BIGINT NOT NULL,
    persona_id BIGINT,
    session_title VARCHAR(256) NOT NULL,
    session_cover VARCHAR(512),
    session_description TEXT,
    script_style VARCHAR(64) DEFAULT 'professional',
    planned_start_time TIMESTAMP,
    planned_end_time TIMESTAMP,
    actual_start_time TIMESTAMP,
    actual_end_time TIMESTAMP,
    status INTEGER NOT NULL DEFAULT 0,
    live_url VARCHAR(512),
    recording_url VARCHAR(512),
    recording_duration INTEGER DEFAULT 0,
    readiness_check VARCHAR(512),
    deleted INTEGER NOT NULL DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_live_session_owner ON live_session(owner_id, status, create_time DESC);
CREATE INDEX idx_live_session_account ON live_session(account_id, status);
CREATE INDEX idx_live_session_time ON live_session(planned_start_time);

-- 2. live_session_product（场次-产品关联）
CREATE TABLE IF NOT EXISTS live_session_product (
    id BIGSERIAL PRIMARY KEY,
    session_id BIGINT NOT NULL,
    product_id BIGINT NOT NULL,
    product_name VARCHAR(256) NOT NULL,
    position INTEGER NOT NULL DEFAULT 1,
    script_source VARCHAR(32) DEFAULT 'session',
    product_script_id BIGINT,
    sale_quantity INTEGER DEFAULT 0,
    revenue DECIMAL(12,2) DEFAULT 0,
    deleted INTEGER NOT NULL DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_live_sp_session ON live_session_product(session_id, position);
CREATE INDEX idx_live_sp_product ON live_session_product(product_id);

-- 3. live_session_script（场次话术）
CREATE TABLE IF NOT EXISTS live_session_script (
    id BIGSERIAL PRIMARY KEY,
    session_id BIGINT NOT NULL,
    product_id BIGINT,
    script_type VARCHAR(32) NOT NULL,
    script_content TEXT NOT NULL,
    style VARCHAR(64),
    sequence_no INTEGER NOT NULL DEFAULT 1,
    execution_time INTEGER,
    executed BOOLEAN DEFAULT false,
    actual_execution_time TIMESTAMP,
    is_ai_generated BOOLEAN DEFAULT false,
    ai_call_log_id BIGINT,
    generation_status VARCHAR(16) DEFAULT 'success',
    violation_checked BOOLEAN DEFAULT false,
    violation_result TEXT,
    viewer_delta INTEGER,
    interaction_delta INTEGER,
    effectiveness_score DECIMAL(5,2),
    deleted INTEGER NOT NULL DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_live_ss_session ON live_session_script(session_id, sequence_no);
CREATE INDEX idx_live_ss_type ON live_session_script(session_id, script_type);
CREATE INDEX idx_live_ss_ai ON live_session_script(ai_call_log_id);
CREATE INDEX idx_live_ss_score ON live_session_script(effectiveness_score DESC);

-- 4. live_session_data（场次汇总数据）
CREATE TABLE IF NOT EXISTS live_session_data (
    id BIGSERIAL PRIMARY KEY,
    session_id BIGINT NOT NULL,
    total_viewers INTEGER DEFAULT 0,
    peak_viewers INTEGER DEFAULT 0,
    avg_watch_duration INTEGER DEFAULT 0,
    new_followers INTEGER DEFAULT 0,
    likes INTEGER DEFAULT 0,
    comments INTEGER DEFAULT 0,
    shares INTEGER DEFAULT 0,
    gift_value DECIMAL(12,2) DEFAULT 0,
    total_gmv DECIMAL(12,2) DEFAULT 0,
    total_orders INTEGER DEFAULT 0,
    conversion_rate DECIMAL(5,2) DEFAULT 0,
    avg_order_value DECIMAL(10,2) DEFAULT 0,
    duration_minutes INTEGER DEFAULT 0,
    ai_analysis TEXT,
    ai_suggestions TEXT,
    ai_review_id BIGINT,
    last_sync_time TIMESTAMP,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE UNIQUE INDEX idx_live_sd_session ON live_session_data(session_id);

-- 5. live_product_data（分产品数据）
CREATE TABLE IF NOT EXISTS live_product_data (
    id BIGSERIAL PRIMARY KEY,
    session_id BIGINT NOT NULL,
    product_id BIGINT NOT NULL,
    product_name VARCHAR(256),
    gmv DECIMAL(12,2) DEFAULT 0,
    orders INTEGER DEFAULT 0,
    click_count INTEGER DEFAULT 0,
    conversion_rate DECIMAL(5,2) DEFAULT 0,
    refund_rate DECIMAL(5,2) DEFAULT 0,
    exposure_count INTEGER DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE UNIQUE INDEX idx_live_pd_sp ON live_product_data(session_id, product_id);
CREATE INDEX idx_live_pd_product ON live_product_data(product_id);

-- 6. live_monitor（实时时序数据）
CREATE TABLE IF NOT EXISTS live_monitor (
    id BIGSERIAL PRIMARY KEY,
    session_id BIGINT NOT NULL,
    record_time TIMESTAMP NOT NULL,
    viewers INTEGER DEFAULT 0,
    total_viewers INTEGER DEFAULT 0,
    likes INTEGER DEFAULT 0,
    comments INTEGER DEFAULT 0,
    shares INTEGER DEFAULT 0,
    new_followers INTEGER DEFAULT 0,
    online_count INTEGER DEFAULT 0,
    gmv DECIMAL(12,2) DEFAULT 0,
    orders INTEGER DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_live_monitor_session ON live_monitor(session_id, record_time);
CREATE INDEX idx_live_monitor_time ON live_monitor(record_time);

-- 7. live_script_template（高效话术入库）
CREATE TABLE IF NOT EXISTS live_script_template (
    id BIGSERIAL PRIMARY KEY,
    template_name VARCHAR(128) NOT NULL,
    script_type VARCHAR(32) NOT NULL,
    category VARCHAR(64),
    content TEXT NOT NULL,
    variables VARCHAR(512),
    effectiveness_score DECIMAL(5,2) DEFAULT 0,
    usage_count INTEGER DEFAULT 0,
    avg_conversion_rate DECIMAL(5,2) DEFAULT 0,
    source_script_id BIGINT,
    source_session_id BIGINT,
    status INTEGER NOT NULL DEFAULT 1,
    deleted INTEGER NOT NULL DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_lst_script_type ON live_script_template(script_type);
CREATE INDEX idx_lst_effectiveness ON live_script_template(effectiveness_score DESC) WHERE deleted = 0;
CREATE INDEX idx_lst_type_category_score ON live_script_template(script_type, category, effectiveness_score DESC) WHERE deleted = 0;
```

## 2.4 字段命名规范

| 规范 | 说明 |
|------|------|
| 表名前缀 | `live_` |
| 主键 | `id`（BIGSERIAL） |
| 外键 | `xxx_id`（如 session_id, product_id） |
| 时间字段 | `xxx_time`（TIMESTAMP） |
| 金额字段 | DECIMAL(12,2) |
| 比率字段 | DECIMAL(5,2)，单位 % |
| 逻辑删除 | `deleted`（INTEGER，0=正常 1=删除） |
| 创建/更新时间 | `create_time` / `update_time` |

## 2.5 数据迁移说明

- 新增表：执行 `sql/live/migration-*.sql` 按顺序执行
- 字段变更：优先使用 ADD COLUMN，避免 ALTER 破坏已有数据
- 跨模块 FK：不使用数据库外键，应用层校验 dy_account、dy_product、dy_persona 存在性
