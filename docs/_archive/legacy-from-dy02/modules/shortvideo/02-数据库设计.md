# 短视频模块 — 数据库设计

> 版本：3.0 | 更新日期：2026-02-26

---

## 2.1 ER 关系总览

```
dy_account (douyin 模块)
    │
    ├─── sv_video（短视频）
    │        │
    │        ├─── sv_video_data（每日数据快照）
    │        │
    │        └─── sv_comment（评论）
    │
    ├─── sv_plan（创作方案）
    │        │
    │        └─── sv_plan_asset（方案素材/版本）
    │
    ├─── sv_viral_video（爆款库）
    │
    └─── sv_publish_time_analysis（发布时间分析）

sv_hot_topic（热点话题，平台级）
sv_category（视频分类，用户级）
sv_video_generation（视频生成任务）
```

## 2.3 初始化 SQL

```sql
-- ============================================
-- shortvideo 模块数据库初始化脚本
-- ============================================

-- 1. sv_video（短视频）
CREATE TABLE IF NOT EXISTS sv_video (
    id BIGSERIAL PRIMARY KEY,
    owner_id BIGINT NOT NULL,
    account_id BIGINT NOT NULL,
    douyin_video_id VARCHAR(128),
    title VARCHAR(512),
    description TEXT,
    cover_url VARCHAR(512),
    video_url VARCHAR(512),
    duration INTEGER DEFAULT 0,
    tags VARCHAR(512),
    category_id BIGINT,
    publish_time TIMESTAMP,
    view_count BIGINT DEFAULT 0,
    like_count INTEGER DEFAULT 0,
    comment_count INTEGER DEFAULT 0,
    share_count INTEGER DEFAULT 0,
    favorite_count INTEGER DEFAULT 0,
    is_viral BOOLEAN DEFAULT false,
    ai_call_log_id BIGINT,
    ai_generated BOOLEAN DEFAULT false,
    plan_id BIGINT,
    sync_status VARCHAR(16) DEFAULT 'synced',
    last_sync_time TIMESTAMP,
    deleted INTEGER NOT NULL DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_sv_video_owner ON sv_video(owner_id, publish_time DESC);
CREATE INDEX idx_sv_video_account ON sv_video(account_id, publish_time DESC);
CREATE UNIQUE INDEX idx_sv_video_douyin ON sv_video(douyin_video_id);
CREATE INDEX idx_sv_video_viral ON sv_video(is_viral, view_count DESC);
CREATE INDEX idx_sv_video_ai ON sv_video(ai_call_log_id);
CREATE INDEX idx_sv_video_category ON sv_video(category_id);

-- 2. sv_video_data（每日数据快照）
CREATE TABLE IF NOT EXISTS sv_video_data (
    id BIGSERIAL PRIMARY KEY,
    video_id BIGINT NOT NULL,
    snapshot_date DATE NOT NULL,
    view_count BIGINT DEFAULT 0,
    like_count INTEGER DEFAULT 0,
    comment_count INTEGER DEFAULT 0,
    share_count INTEGER DEFAULT 0,
    favorite_count INTEGER DEFAULT 0,
    new_followers INTEGER DEFAULT 0,
    view_delta BIGINT DEFAULT 0,
    like_delta INTEGER DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE UNIQUE INDEX idx_sv_vd_video_date ON sv_video_data(video_id, snapshot_date);
CREATE INDEX idx_sv_vd_date ON sv_video_data(snapshot_date);

-- 3. sv_comment（评论）
CREATE TABLE IF NOT EXISTS sv_comment (
    id BIGSERIAL PRIMARY KEY,
    video_id BIGINT NOT NULL,
    douyin_comment_id VARCHAR(128),
    content TEXT NOT NULL,
    author_name VARCHAR(128),
    author_avatar VARCHAR(512),
    like_count INTEGER DEFAULT 0,
    reply_count INTEGER DEFAULT 0,
    sentiment VARCHAR(16),
    sentiment_score DECIMAL(3,2),
    comment_time TIMESTAMP,
    deleted INTEGER NOT NULL DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_sv_comment_video ON sv_comment(video_id, comment_time DESC);
CREATE UNIQUE INDEX idx_sv_comment_douyin ON sv_comment(douyin_comment_id);
CREATE INDEX idx_sv_comment_sentiment ON sv_comment(video_id, sentiment);

-- 4. sv_plan（创作方案）
CREATE TABLE IF NOT EXISTS sv_plan (
    id BIGSERIAL PRIMARY KEY,
    owner_id BIGINT NOT NULL,
    account_id BIGINT NOT NULL,
    persona_id BIGINT,
    plan_type VARCHAR(32) NOT NULL,
    title VARCHAR(256) NOT NULL,
    description TEXT,
    content TEXT,
    script_content TEXT,
    source_video_id BIGINT,
    source_topic_id BIGINT,
    ai_call_log_id BIGINT,
    linked_video_id BIGINT,
    content_effect VARCHAR(32),
    effect_score DECIMAL(8,2),
    status VARCHAR(16) NOT NULL DEFAULT 'draft',
    planned_publish_time TIMESTAMP,
    actual_publish_time TIMESTAMP,
    deleted INTEGER NOT NULL DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_sv_plan_owner ON sv_plan(owner_id, status, create_time DESC);
CREATE INDEX idx_sv_plan_account ON sv_plan(account_id, plan_type);
CREATE INDEX idx_sv_plan_ai ON sv_plan(ai_call_log_id);
CREATE INDEX idx_sv_plan_video ON sv_plan(linked_video_id);
CREATE INDEX idx_sv_plan_publish ON sv_plan(planned_publish_time);

-- 5. sv_plan_asset（方案素材/版本）
CREATE TABLE IF NOT EXISTS sv_plan_asset (
    id BIGSERIAL PRIMARY KEY,
    plan_id BIGINT NOT NULL,
    asset_type VARCHAR(32) NOT NULL,
    content TEXT,
    file_url VARCHAR(512),
    file_size BIGINT DEFAULT 0,
    duration INTEGER DEFAULT 0,
    version INTEGER NOT NULL DEFAULT 1,
    generation_task_id BIGINT,
    metadata TEXT,
    deleted INTEGER NOT NULL DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_sv_pa_plan ON sv_plan_asset(plan_id, asset_type, version DESC);
CREATE INDEX idx_sv_pa_task ON sv_plan_asset(generation_task_id);

-- 6. sv_viral_video（爆款库）
CREATE TABLE IF NOT EXISTS sv_viral_video (
    id BIGSERIAL PRIMARY KEY,
    owner_id BIGINT,
    account_id BIGINT,
    video_id BIGINT,
    source VARCHAR(32) NOT NULL DEFAULT 'manual',
    title VARCHAR(512) NOT NULL,
    description TEXT,
    cover_url VARCHAR(512),
    video_url VARCHAR(512),
    author_name VARCHAR(128),
    industry VARCHAR(64),
    view_count BIGINT DEFAULT 0,
    like_count INTEGER DEFAULT 0,
    viral_score DECIMAL(5,2) DEFAULT 0,
    success_factors TEXT,
    replicable_methods TEXT,
    analysis_report_id BIGINT,
    detected_time TIMESTAMP,
    tags VARCHAR(512),
    deleted INTEGER NOT NULL DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_sv_vv_industry ON sv_viral_video(industry, view_count DESC);
CREATE UNIQUE INDEX idx_sv_vv_video ON sv_viral_video(video_id);
CREATE INDEX idx_sv_vv_source ON sv_viral_video(source, create_time DESC);
CREATE INDEX idx_sv_vv_score ON sv_viral_video(viral_score DESC);

-- 7. sv_hot_topic（热点话题）
CREATE TABLE IF NOT EXISTS sv_hot_topic (
    id BIGSERIAL PRIMARY KEY,
    source VARCHAR(32) NOT NULL DEFAULT 'manual',
    douyin_hot_id VARCHAR(128),
    title VARCHAR(256) NOT NULL,
    description TEXT,
    heat_score DECIMAL(10,2) DEFAULT 0,
    category VARCHAR(64),
    related_tags VARCHAR(512),
    status VARCHAR(16) NOT NULL DEFAULT 'active',
    expiry_time TIMESTAMP,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_sv_ht_status ON sv_hot_topic(status, heat_score DESC);
CREATE UNIQUE INDEX idx_sv_ht_douyin ON sv_hot_topic(douyin_hot_id);
CREATE INDEX idx_sv_ht_expiry ON sv_hot_topic(expiry_time);
CREATE INDEX idx_sv_ht_category ON sv_hot_topic(category, status);

-- 8. sv_category（视频分类）
CREATE TABLE IF NOT EXISTS sv_category (
    id BIGSERIAL PRIMARY KEY,
    owner_id BIGINT NOT NULL,
    name VARCHAR(64) NOT NULL,
    description VARCHAR(256),
    sort_order INTEGER DEFAULT 0,
    deleted INTEGER NOT NULL DEFAULT 0,
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE UNIQUE INDEX idx_sv_cat_owner_name ON sv_category(owner_id, name);
CREATE INDEX idx_sv_cat_sort ON sv_category(owner_id, sort_order);

-- 9. sv_video_generation（视频生成任务）
CREATE TABLE IF NOT EXISTS sv_video_generation (
    id BIGSERIAL PRIMARY KEY,
    owner_id BIGINT NOT NULL,
    plan_id BIGINT,
    task_type VARCHAR(32) NOT NULL,
    input_data TEXT,
    output_url VARCHAR(512),
    output_metadata TEXT,
```

## 2.4 字段命名规范

| 规范 | 说明 |
|------|------|
| 表名前缀 | `sv_` |
| 主键 | `id`（BIGSERIAL） |
| 外键 | `xxx_id`（如 video_id, plan_id） |
| 时间字段 | `xxx_time`（TIMESTAMP） |
| 日期字段 | `xxx_date`（DATE） |
| 金额字段 | DECIMAL(12,2) |
| 比率字段 | DECIMAL(5,2)，单位 % |
| 逻辑删除 | `deleted`（INTEGER，0=正常 1=删除） |
| 创建/更新时间 | `create_time` / `update_time` |
